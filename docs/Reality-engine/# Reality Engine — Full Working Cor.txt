# Reality Engine — Full Working Core

Complete, expandable implementation scaffold. Every observer follows the same interface. Every event flows through the same bus. Adding a new capability = adding one module.

### `src/core/event-bus.ts` — The Heartbeat

```typescript
import type { RealityEvent, RealityEventType, ObserverId } from "./types";
import { createEventId } from "./identity";

type EventFilter = {
  types?: RealityEventType[];
  sources?: ObserverId[];
  subjectPattern?: string;
};

type EventHandler = (event: RealityEvent) => void | Promise<void>;

interface Subscription {
  id: string;
  filter: EventFilter;
  handler: EventHandler;
}

export class EventBus {
  private subscriptions: Subscription[] = [];
  private sequence = 0;
  private eventLog: RealityEvent[] = [];
  private maxLogSize = 10_000;

  subscribe(filter: EventFilter, handler: EventHandler): () => void {
    const id = `sub_${Date.now()}_${Math.random().toString(36).slice(2)}`;
    const sub: Subscription = { id, filter, handler };
    this.subscriptions.push(sub);

    // Return unsubscribe function
    return () => {
      this.subscriptions = this.subscriptions.filter((s) => s.id !== id);
    };
  }

  async emit(
    partial: Omit<RealityEvent, "eventId" | "observedAt" | "seq">
  ): Promise<RealityEvent> {
    const event: RealityEvent = {
      ...partial,
      eventId: createEventId(),
      observedAt: new Date().toISOString(),
      seq: ++this.sequence,
    };

    // Append to log (bounded)
    this.eventLog.push(event);
    if (this.eventLog.length > this.maxLogSize) {
      this.eventLog = this.eventLog.slice(-this.maxLogSize);
    }

    // Dispatch to matching subscribers
    const promises = this.subscriptions
      .filter((sub) => this.matchesFilter(event, sub.filter))
      .map((sub) => {
        try {
          return Promise.resolve(sub.handler(event));
        } catch (err) {
          console.error(
            `[EventBus] Handler error for ${sub.id}:`,
            err
          );
          return Promise.resolve();
        }
      });

    await Promise.allSettled(promises);
    return event;
  }

  getRecentEvents(limit = 100): RealityEvent[] {
    return this.eventLog.slice(-limit);
  }

  getSequence(): number {
    return this.sequence;
  }

  private matchesFilter(event: RealityEvent, filter: EventFilter): boolean {
    if (filter.types && !filter.types.includes(event.type)) return false;
    if (filter.sources && !filter.sources.includes(event.source)) return false;
    if (filter.subjectPattern) {
      const re = new RegExp(filter.subjectPattern.replace(/\*/g, ".*"));
      if (!re.test(event.subjectId)) return false;
    }
    return true;
  }
}

  if (!_bus) _bus = new EventBus();
  return _bus;
}

  _bus = null;
```

### `src/observers/interface.ts` — The Observer Contract

```typescript
import type {
  Observation,
  ObserverId,
  BasisRef,
  RealityEvent,
} from "../core/types";

/**
 * Every observer must implement this interface.
 * Adding a new observer = implementing this + registering it.
 */
export interface Observer {
  /** Unique observer identifier */
  readonly id: ObserverId;

  /** Human-readable name */
  readonly name: string;

  /** Initialize the observer (start watchers, etc.) */
  init(): Promise<void>;

  /** Perform a fresh observation of the specified scope */
  observe(options?: ObserveOptions): Promise<Observation[]>;

  /** Get the current state (from cache if fresh enough) */
  getStatus(): Promise<ObserverStatus>;

  /** Shut down the observer (stop watchers, close handles) */
  dispose(): Promise<void>;
}

export interface ObserveOptions {
  /** Force fresh observation, bypass cache */
  fresh?: boolean;
  /** Specific subjects to observe (if omitted, observe all) */
  subjects?: string[];
  /** Maximum observation time in ms */
  timeoutMs?: number;
}

export interface ObserverStatus {
  observer: ObserverId;
  state: "HEALTHY" | "DEGRADED" | "FAILED" | "DISABLED";
  lastSuccess?: string;
  lastFailure?: string;
  lastError?: string;
  observationCount: number;
  errorCount24h: number;
}

/**
 * Base class for observers — provides common patterns.
 * Subclasses implement `collectRaw()` and `classify()`.
 */
export abstract class BaseObserver implements Observer {
  abstract readonly id: ObserverId;
  abstract readonly name: string;

  protected observations: Map<string, Observation> = new Map();
  protected status: ObserverStatus = {
    observer: "reality-core",
    state: "DISABLED",
    observationCount: 0,
    errorCount24h: 0,
  };

  abstract init(): Promise<void>;
  abstract dispose(): Promise<void>;

  async observe(options?: ObserveOptions): Promise<Observation[]> {
    const start = Date.now();
    try {
      const raw = await this.collectRaw(options);
      const observations = await this.classify(raw, options);

      for (const obs of observations) {
        const existing = this.observations.get(obs.subjectId);
        if (existing) {
          obs.previousObservationId = existing.id;
        }
        this.observations.set(obs.subjectId, obs);
      }

      this.status = {
        ...this.status,
        state: "HEALTHY",
        lastSuccess: new Date().toISOString(),
        observationCount: this.status.observationCount + 1,
      };

      return observations;
    } catch (err) {
      this.status = {
        ...this.status,
        state: "DEGRADED",
        lastFailure: new Date().toISOString(),
        lastError:
          err instanceof Error ? err.message : String(err),
        errorCount24h: this.status.errorCount24h + 1,
      };

      // Return cached observations if available and fresh enough
      if (this.observations.size > 0) {
        return Array.from(this.observations.values()).map((obs) => ({
          ...obs,
          freshness: "STALE" as const,
          error: {
            code: "ENG_OBSERVER_FAILED" as const,
            message:
              err instanceof Error ? err.message : String(err),
            retryable: true,
          },
        }));
      }

      throw err;
    }
  }

  async getStatus(): Promise<ObserverStatus> {
    return { ...this.status };
  }

  /** Collect raw data from the machine. Observer-specific. */
  protected abstract collectRaw(
    options?: ObserveOptions
  ): Promise<unknown>;

  /** Classify raw data into observations. Observer-specific. */
  protected abstract classify(
    raw: unknown,
    options?: ObserveOptions
  ): Promise<Observation[]>;
}
```

```typescript
import { BaseObserver, type ObserveOptions } from "../interface";
import type {
  Observation,
  ObserverId,
  BasisRef,
  RepositorySnapshot,
  ChangedFile,
} from "../../core/types";
import { createObservationId } from "../../core/identity";
import { canonicalSerialize } from "../../core/serialization";

export class GitObserver extends BaseObserver {
  readonly id: ObserverId = "reality-git";
  readonly name = "Git Repository Observer";

  private lastHead: string | null = null;
  private lastRemoteMain: string | null = null;
  private lastDirty: boolean | null = null;

  async init(): Promise<void> {
    // Verify git is available
    try {
      await execGit(["--version"]);
      this.status.state = "HEALTHY";
    } catch {
      this.status.state = "FAILED";
      this.status.lastError = "Git CLI not available";

  async dispose(): Promise<void> {
    // Nothing to clean up for git observer
    this.status.state = "DISABLED";

  protected async collectRaw(
    _options?: ObserveOptions
  ): Promise<GitRawState> {
    const root = await execGit(["rev-parse", "--show-toplevel"]);
    const head = await execGit(["rev-parse", "HEAD"]).catch(
      () => "UNBORN"
    );
    const branch = await execGit([
      "branch",
      "--show-current",
    ]).catch(() => "");
    const isDetached = branch === "";

    const localMain = await execGit([
      "rev-parse",
      "refs/heads/main",
    ]).catch(() => "");
    const remoteMain = await execGit([
      "rev-parse",
      "refs/remotes/origin/main",
    ]).catch(() => "");

    const mergeBase =
      head !== "UNBORN" && remoteMain
        ? await execGit(["merge-base", "HEAD", remoteMain]).catch(
            () => ""
          )
        : "";

    const divergenceRaw = await execGit([
      "rev-list",
      "--left-right",
      "--count",
      `HEAD...${remoteMain || "HEAD"}`,
    ]).catch(() => "0\t0");
    const [aheadStr, behindStr] = divergenceRaw.split("\t");

    const statusRaw = await execGit([
      "status",
      "--porcelain=v2",
      "-z",

    const gitVersion = await execGit(["--version"]);

      root: root.trim(),
      head: head.trim(),
      branch: branch.trim(),
      isDetached,
      localMain: localMain.trim(),
      remoteMain: remoteMain.trim(),
      mergeBase: mergeBase.trim(),
      ahead: parseInt(aheadStr, 10) || 0,
      behind: parseInt(behindStr, 10) || 0,
      statusEntries,
      remoteUrls,
      gitVersion: gitVersion.trim(),
    };
  }

  protected async classify(
    raw: unknown,
    _options?: ObserveOptions
  ): Promise<Observation[]> {
    const state = raw as GitRawState;
    const now = new Date().toISOString();
    const bus = getEventBus();

    const basis: BasisRef[] = [
      {
        source: "git",
        ref: ".git/HEAD",
        digest: state.head,
        confirmedAt: now,
      },
      {
        source: "git",
        ref: "refs/remotes/origin/main",
        revision: state.remoteMain,
        confirmedAt: now,
      },
    ];

    const changedFiles: ChangedFile[] = state.statusEntries.map(
      (entry) => ({
        path: entry.path,
        status: entry.status as ChangedFile["status"],
        oldPath: entry.oldPath,
        staged: entry.staged,
      })
    );

    const isDirty = changedFiles.length > 0;

    // Emit events for state transitions
    if (this.lastHead && this.lastHead !== state.head) {
      await bus.emit({
        type: "GIT_HEAD_CHANGED",
        occurredAt: now,
        source: this.id,
        subjectId: `repo:${state.root}`,
        payload: { from: this.lastHead, to: state.head },
        basis,
      });
    }

    if (
      this.lastRemoteMain &&
      this.lastRemoteMain !== state.remoteMain
    ) {
      await bus.emit({
        type: "GIT_MAIN_ADVANCED",
        occurredAt: now,
        source: this.id,
        subjectId: `repo:${state.root}`,
        payload: {
          from: this.lastRemoteMain,
          to: state.remoteMain,
        },
        basis,
      });
    }

    if (this.lastDirty !== null && this.lastDirty !== isDirty) {
      await bus.emit({
        type: isDirty
          ? "GIT_WORKTREE_DIRTIED"
          : "GIT_WORKTREE_CLEANED",
        occurredAt: now,
        source: this.id,
        subjectId: `repo:${state.root}`,
        payload: { dirty: isDirty, fileCount: changedFiles.length },
        basis,
      });
    }

    // Update tracking state
    this.lastHead = state.head;
    this.lastRemoteMain = state.remoteMain;
    this.lastDirty = isDirty;

    const snapshot: RepositorySnapshot = {
      repositoryRoot: state.root,
      remoteUrls: state.remoteUrls,
      currentBranch: state.branch,
      head: state.head,
      localMain: state.localMain,
      remoteMain: state.remoteMain,
      mergeBase: state.mergeBase,
      ahead: state.ahead,
      behind: state.behind,
      isClean: !isDirty,
      isDetached: state.isDetached,
      isDiverged: state.ahead > 0 && state.behind > 0,
      canFastForward: state.ahead === 0 && state.behind > 0,
      changedFiles,
      untrackedFiles: state.statusEntries
        .filter((e) => e.status === "ADDED" && !e.staged)
        .map((e) => e.path),
      gitVersion: state.gitVersion,
      activeWriters: [], // Populated by contention detector
    };

    return [
      {
        id: createObservationId(),
        kind: "git.repository.state",
        observedAt: now,
        source: this.id,
        subjectId: `repo:${state.root}`,
        status: "OBSERVED",
        freshness: "CURRENT",

        value: snapshot,
        scope: {

          exclusions: [],
          windowMs: 0,
        },
        schemaVersion: "reality/1.0.0",
      },
    ];
  }
}

interface GitRawState {
  root: string;
  head: string;
  branch: string;
  isDetached: boolean;
  localMain: string;
  remoteMain: string;
  mergeBase: string;
  ahead: number;
  behind: number;
  statusEntries: GitStatusEntry[];
  remoteUrls: string[];
  gitVersion: string;
}

function parseGitStatus(raw: string): GitStatusEntry[] {
  if (!raw) return [];
  const entries: GitStatusEntry[] = [];
  // porcelain v2 with -z: NUL-delimited
  const parts = raw.split("\0").filter(Boolean);
  let i = 0;
  while (i < parts.length) {
    const line = parts[i];
    if (line.startsWith("1 ") || line.startsWith("2 ")) {
      // Ordinary or rename entry
      const tokens = line.split(" ");
      const xy = tokens[1]; // XY status
      const path =
        line.startsWith("2 ") ? parts[i + 1] : tokens[tokens.length - 1];
      entries.push({
        path: path || "",
        status: statusFromXY(xy),
        oldPath: line.startsWith("2 ") ? tokens[tokens.length - 1] : undefined,
        staged: xy[0] !== ".",
      });
      i += line.startsWith("2 ") ? 2 : 1;
    } else if (line.startsWith("? ")) {
      entries.push({
        path: line.slice(2),
        status: "ADDED",
        staged: false,
      });
      i++;
    } else if (line.startsWith("u ")) {
      const tokens = line.split(" ");
      entries.push({
        path: tokens[tokens.length - 1],
        status: "UNMERGED",
        staged: false,
      });
      i++;
    } else {
      i++;
    }
  }
  return entries;
}

function statusFromXY(
  xy: string
): ChangedFile["status"] {
  if (xy.includes("M")) return "MODIFIED";
  if (xy.includes("A")) return "ADDED";
  if (xy.includes("D")) return "DELETED";
  if (xy.includes("R")) return "RENAMED";
  if (xy.includes("C")) return "COPIED";
  return "MODIFIED";
}

function parseRemoteUrls(raw: string): string[] {
  const urls: string[] = [];
  const lines = raw.split("\n");
  for (const line of lines) {
    if (line.includes("(fetch)")) {
      const parts = line.split(/\s+/);
      if (parts.length >= 2) urls.push(parts[1]);
    }
  }
  return urls;
}

function hashPath(path: string): string {
  // Simple hash for workspace ID
  let hash = 0;
  const normalized = path.toLowerCase().replace(/\\/g, "/");
  for (let i = 0; i < normalized.length; i++) {
    const ch = normalized.charCodeAt(i);
    hash = ((hash << 5) - hash + ch) | 0;
  }
  return Math.abs(hash).toString(16).padStart(8, "0");
}
```

```typescript
let _gitPath: string | null = null;

async function findGit(): Promise<string> {
  if (_gitPath) return _gitPath;

  // Try common locations on Windows
  const candidates = [
    "git",
    "C:\\Program Files\\Git\\bin\\git.exe",
    "C:\\Program Files (x86)\\Git\\bin\\git.exe",
    `${process.env.LOCALAPPDATA}\\Programs\\Git\\bin\\git.exe`,
  ];

  for (const candidate of candidates) {
    try {
      const proc = spawn([candidate, "--version"], {
        stdout: "pipe",
        stderr: "pipe",
      });
      await proc.exited;
      if (proc.exitCode === 0) {
        _gitPath = candidate;
        return candidate;
      }
    } catch {
      // Try next
    }
  }

  throw new Error(
    "Git CLI not found. Install Git or add it to PATH."
  );
}

export async function execGit(
  args: string[],
  cwd?: string
): Promise<string> {
  const git = await findGit();
  const proc = spawn([git, ...args], {
    cwd: cwd || process.cwd(),
    stdout: "pipe",
    stderr: "pipe",
    env: {
      ...process.env,
      GIT_TERMINAL_PROMPT: "0",
    },
  });

  const stdout = await new Response(proc.stdout).text();
  const stderr = await new Response(proc.stderr).text();

  await proc.exited;

  if (proc.exitCode !== 0) {
    throw new GitError(
      `git ${args.join(" ")} failed (exit ${proc.exitCode}): ${stderr.trim()}`,
      args,
      proc.exitCode
    );
  }

  return stdout.trimEnd();
}

export class GitError extends Error {
  constructor(
    message: string,
    public args: string[],
    public exitCode: number
  ) {
    super(message);
    this.name = "GitError";
  }
}
```

```typescript
      agentAssumedMainCommit: this.agentAssumedMainCommit,
      currentMainCommit: currentMain,
      mainAdvanced: advanced,
      commitsAdvanced: advanced ? 1 : 0, // Simplified; use rev-list for count
      driftStatus: advanced ? "STALE_MAIN_BASIS" : "NO_DRIFT",
      detectedAt: new Date().toISOString(),
    };
  }

  private async getCollisions(): Promise<IdentityCollision[]> {
    return this.storage.getCollisions();
  }

  private async computeBasisDigest(
    observations: Observation[]
  ): Promise<string> {
    const bases = observations.flatMap((o) => o.basis);
    const sorted = bases.sort((a, b) =>
      `${a.source}:${a.ref}`.localeCompare(`${b.source}:${b.ref}`)
    );
    const serialized = canonicalSerialize(sorted);
    // Simple hash — use crypto in production
    let hash = 0;
    for (let i = 0; i < serialized.length; i++) {
      hash = ((hash << 5) - hash + serialized.charCodeAt(i)) | 0;
    }
    return Math.abs(hash).toString(16).padStart(16, "0");
  }

  private computeOverallFreshness(
    observations: Observation[]
  ): "CURRENT" | "STALE" | "UNRESOLVABLE" {
    if (observations.some((o) => o.freshness === "UNRESOLVABLE"))
      return "UNRESOLVABLE";
    if (observations.some((o) => o.freshness === "STALE"))
      return "STALE";
    return "CURRENT";
  }

  private getUnavailable(): Array<{
    kind: string;
    reason: string;
    code: any;
  }> {
    // Collect from degraded observers
    return [];
  }

  private async isCommitOnMain(sha: string): Promise<boolean> {
    try {
      const result = await import("../observers/git/commands").then(
        (m) =>
          m.execGit([
            "branch",
            "--contains",
            sha,
            "--list",
            "main",
          ])
      );
      return result.includes("main");
    } catch {
      return false;
    }
  }

  private async getHealth(): Promise<any> {
    return this.health();
  }

  // Extract helpers — pull typed snapshots from observations
  private extractWorkspace(obs: Observation[]): any {
    const gitObs = obs.find((o) =>
      o.kind.startsWith("git.repository")
    );
    if (!gitObs) return { workspaceId: "unknown" };
    const repo = gitObs.value as any;
    return {
      workspaceId: `workspace:${repo.repositoryRoot}`,
      rootPath: repo.repositoryRoot,
      branch: repo.currentBranch,
      head: repo.head,
      dirtyState: repo.isClean ? "CLEAN" : "DIRTY",
      lockState: "UNLOCKED",
      lastObservation: gitObs.observedAt,
    };
  }

  private extractRepository(obs: Observation[]): any {
    const gitObs = obs.find((o) =>
      o.kind.startsWith("git.repository")
    );
    return gitObs?.value || {};
  }

  private extractFilesystem(obs: Observation[]): any {
    return { observedAt: new Date().toISOString(), files: [], directories: [] };
  }

  private extractProcesses(obs: Observation[]): any {
    return { observedAt: new Date().toISOString(), processes: [] };
  }

  private extractOpenCode(obs: Observation[]): any {
    return { sessions: [], detectedAt: new Date().toISOString(), integrationMethod: "process-observation" };
  }

  private extractToolchain(obs: Observation[]): any {
    return { tools: [], observedAt: new Date().toISOString() };
  }

  private extractVerification(obs: Observation[]): any {

  }
```

```typescript
export interface LoopCallbacks {
  fast: () => Promise<void>;
  medium: () => Promise<void>;
  slow: () => Promise<void>;
}

export class LoopScheduler {
  private timers: NodeJS.Timeout[] = [];
  private running = false;

  constructor(
    private intervals: {
      fastMs: number;
      mediumMs: number;
      slowMs: number;
    }
  ) {}

  start(callbacks: LoopCallbacks): void {
    if (this.running) return;
    this.running = true;

    const safeCall = async (
      name: string,
      fn: () => Promise<void>
    ) => {
      try {
        await fn();
      } catch (err) {
        console.error(`[LoopScheduler] ${name} loop error:`, err);
      }
    };

    this.timers.push(
      setInterval(
        () => safeCall("fast", callbacks.fast),
        this.intervals.fastMs
      ),
      setInterval(
        () => safeCall("medium", callbacks.medium),
        this.intervals.mediumMs
      ),
      setInterval(
        () => safeCall("slow", callbacks.slow),
        this.intervals.slowMs
      )
    );

    // Run immediately
    safeCall("fast-init", callbacks.fast);
    safeCall("medium-init", callbacks.medium);
    safeCall("slow-init", callbacks.slow);
  }

  stop(): void {
    this.running = false;
    for (const timer of this.timers) {
      clearInterval(timer);
    }
    this.timers = [];
  }
}
```

```typescript
  const flags: Record<string, any> = {};
  for (let i = 0; i < args.length; i++) {
    const arg = args[i];

      if (args[i + 1] && !args[i + 1].startsWith("--")) {
        flags[key] = args[i + 1];
        i++;
      } else {
        flags[key] = true;
      }
    }
  }
```

### `src/core/serialization.ts` — Canonical JSON

```typescript
/**
 * Canonical JSON serialization for Reality Engine.
 * - Alphabetical key ordering
 * - Stable array ordering (sorted by serialized element)
 * - Omitted absent fields (never "null")
 * - UTF-8 encoding
 * - ISO 8601 dates with timezone offset
 * - NFC-normalized strings
 */
export function canonicalSerialize(value: unknown): string {
  return JSON.stringify(canonicalSort(value), null, 2);
}

function canonicalSort(value: unknown): unknown {
  if (value === null || value === undefined) return undefined;

  if (Array.isArray(value)) {
    const sorted = value
      .map(canonicalSort)
      .filter((v) => v !== undefined);
    // Sort arrays by their serialized form for determinism
    return sorted.sort((a, b) =>
      JSON.stringify(a).localeCompare(JSON.stringify(b))
    );
  }

  if (typeof value === "object") {
    const obj = value as Record<string, unknown>;
    const sorted: Record<string, unknown> = {};
    const keys = Object.keys(obj).sort();
    for (const key of keys) {
      const sortedValue = canonicalSort(obj[key]);
      if (sortedValue !== undefined) {
        sorted[key] = sortedValue;
      }
    }
    return sorted;
  }

  if (typeof value === "string") {
    return value.normalize("NFC");
  }

  return value;
}
```

```typescript
      ...userConfig,
      workspaceRoot,
    }) as RealityConfig;
  } catch {
    return { ...DEFAULT_CONFIG, workspaceRoot };
  }
}

function deepMerge(base: any, override: any): any {
  const result = { ...base };
  for (const key of Object.keys(override)) {
    if (
      typeof override[key] === "object" &&
      override[key] !== null &&
      !Array.isArray(override[key]) &&
      typeof base[key] === "object" &&
      base[key] !== null
    ) {
      result[key] = deepMerge(base[key], override[key]);
    } else {
      result[key] = override[key];
    }
  }
  return result;
}
```

## How to Expand

**Adding a new observer:**

1. Create `src/observers/<name>/index.ts`
2. Extend `BaseObserver`
3. Implement `collectRaw()` and `classify()`
4. Register in `src/observers/registry.ts`
5. Add scope mapping in `getForScope()`

**Adding a new event type:**

1. Add to `RealityEventType` union in `types.ts`
2. Emit it from the relevant observer
3. Subscribe in the engine if it triggers cross-cutting behavior

**Adding a new CLI command:**

1. Create `src/cli/commands/<name>.ts`
2. Add case in `src/cli/index.ts`
3. Add to `COMMANDS` help map

**Adding a new CFA profile:**

1. Add to `cfaProfiles` in config
2. The `reality for <cfa-id>` command filters automatically
